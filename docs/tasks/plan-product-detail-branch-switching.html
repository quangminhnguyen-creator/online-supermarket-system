  • Step 5: Tạo state model và fetch tối thiểu

    type ProductState =
  | { kind: 'loading' }
  | { kind: 'ready'; data: ProductDetailDto }
  | { kind: 'not-found' }
  | { kind: 'error'; message: string }

export function ProductDetailPage() {
  const { id } = useParams<{ id: string }>()
  const [searchParams] = useSearchParams()
  const branchId = searchParams.get('branchId') || undefined
  const [productState, setProductState] = useState<ProductState>({ kind: 'loading' })

  useEffect(() => {
    const controller = new AbortController()
    if (!id) { setProductState({ kind: 'not-found' }); return () => controller.abort() }
    setProductState({ kind: 'loading' })
    catalogApi.getProductById(id, branchId, controller.signal)
      .then((data) => setProductState({ kind: 'ready', data }))
      .catch((error: unknown) => {
        if (controller.signal.aborted) return
        if (error instanceof ApiError && error.status === 404) {
          setProductState({ kind: 'not-found' }); return
        }
        setProductState({ kind: 'error', message: 'Không thể tải chi tiết sản phẩm.' })
      })
    return () => controller.abort()
  }, [id, branchId])
}

const [searchParams, setSearchParams] = useSearchParams()
const [branchState, setBranchState] = useState<BranchState>({ kind: 'loading' })

function handleBranchChange(event: ChangeEvent<HTMLSelectElement>) {
  const next = new URLSearchParams(searchParams)
  const nextBranchId = event.target.value
  if (nextBranchId) next.set('branchId', nextBranchId)
  else next.delete('branchId')
  setSearchParams(next)
}

  {branchState.kind === 'ready' && branchState.data.map((branch) => (

    function getAvailabilityView(product: ProductDetailDto, branchId?: string) {
  if (!branchId) {
    return { price: product.basePrice, tone: 'neutral', message: 'Chọn kho để xem giá và tồn kho' }
  }
  if (!product.branchInventory) {
    return { price: product.basePrice, tone: 'unavailable', message: 'Sản phẩm không có tại kho này' }
  }
  if (product.branchInventory.availableQuantity === 0) {
    return { price: product.branchInventory.sellingPrice, tone: 'unavailable', message: 'Tạm hết hàng tại kho này' }
  }
  return {
    price: product.branchInventory.sellingPrice,
    tone: 'available',
    message: `Còn ${product.branchInventory.availableQuantity} sản phẩm tại kho`,
  }
}
